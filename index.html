async function loadCampaigns(loadMore) { const from = document.getElementById('date-from').value; const to = document.getElementById('date-to').value; if (!from || !to) { showStatus('campaign-status', 'Please select a start and end date.', true); return; } if (!loadMore) { campaignOffset = 0; allFetchedCampaigns = []; currentFromTs = from; currentToTs = to; const sel = document.getElementById('campaign-select'); sel.innerHTML = ''; document.getElementById('load-more-row').style.display = 'none'; } const btn = loadMore ? document.getElementById('load-more-btn') : document.getElementById('load-btn'); btn.disabled = true; showStatus('campaign-status', loadMore ? 'Loading more…' : 'Loading campaigns…'); try { const path = 'campaigns?limit=' + CAMPAIGN_PAGE_SIZE + '&offset=' + campaignOffset + '&orders[sdate]=DESC&filters[status]=5'; const data = await acFetch(path); const campaigns = data.campaigns || []; // Filter by date string comparison to avoid timezone issues // AC sdate format: "YYYY-MM-DD HH:MM:SS", substring(0,10) = "YYYY-MM-DD" const filtered = campaigns.filter(c => { if (!c.sdate) return false; const cDate = c.sdate.substring(0, 10); return cDate >= currentFromTs && cDate <= currentToTs; }); filtered.forEach(c => { campaignData[c.id] = c; allFetchedCampaigns.push(c); const opt = document.createElement('option'); opt.value = c.id; const d = c.sdate ? new Date(c.sdate).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' }) : ''; opt.textContent = (d ? d + ' — ' : '') + c.name; document.getElementById('campaign-select').appendChild(opt); }); document.getElementById('campaign-select').onchange = () => onCampaignSelect(document.getElementById('campaign-select').value); campaignOffset += campaigns.length; const oldestDate = campaigns.length > 0 ? campaigns[campaigns.length - 1].sdate.substring(0, 10) : null; const hasMore = campaigns.length === CAMPAIGN_PAGE_SIZE && oldestDate && oldestDate >= currentFromTs; const total = allFetchedCampaigns.length; if (total === 0 && !loadMore) { showStatus('campaign-status', 'No sent campaigns found in that date range.', true); return; } if (hasMore) { document.getElementById('load-more-row').style.display = 'block'; showStatus('campaign-status', total + ' campaigns found in range — more may be available.'); } else { document.getElementById('load-more-row').style.display = 'none'; showStatus('campaign-status', total + ' campaign' + (total !== 1 ? 's' : '') + ' found.'); } } catch(e) { showStatus('campaign-status', e.message, true); } finally { btn.disabled = false; } }
